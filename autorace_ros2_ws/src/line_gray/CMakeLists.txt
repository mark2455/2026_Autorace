cmake_minimum_required(VERSION 3.8)
project(gray_detector)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(autorace_ros2_compat REQUIRED)
find_package(coss_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED)
add_executable(gray_detector_node src/gray.cpp)
target_include_directories(gray_detector_node PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(gray_detector_node autorace_ros2_compat coss_msgs rclcpp sensor_msgs std_msgs cv_bridge)
target_link_libraries(gray_detector_node ${OpenCV_LIBRARIES})
install(TARGETS gray_detector_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
